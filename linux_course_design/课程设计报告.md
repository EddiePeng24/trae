#### **3.3.2 创建用户功能**

```bash
create_user() {
    echo -e "${BLUE}============================================${NC}"
    echo -e "${BLUE}           创建新用户${NC}"
    echo -e "${BLUE}============================================${NC}"

    read -p "请输入要创建的用户名: " username

    # 【逻辑说明】检查用户名是否为空
    if [ -z "$username" ]; then
        echo -e "${RED}[错误] 用户名不能为空！${NC}"
        return 1
    fi

    # 【核心逻辑】使用id命令检查用户是否已存在
    # id命令会查询/etc/passwd文件，如果用户存在返回0，否则非0
    if id "$username" &>/dev/null; then
        echo -e "${RED}[错误] 用户 '$username' 已存在！${NC}"
        write_log "尝试创建用户失败: 用户 $username 已存在"
        return 1
    fi

    # 输入密码并进行一致性校验
    read -s -p "请输入用户密码: " password
    echo ""
    read -s -p "请再次确认密码: " password2
    echo ""

    if [ "$password" != "$password2" ]; then
        echo -e "${RED}[错误] 两次输入的密码不一致！${NC}"
        return 1
    fi

    # 【核心命令】使用useradd创建用户
    # -m 参数：自动创建用户主目录(/home/username)
    # -s /bin/bash：指定默认Shell为bash
    echo -e "${YELLOW}[正在创建用户]...${NC}"
    sudo useradd -m -s /bin/bash "$username"

    if [ $? -eq 0 ]; then
        # 使用chpasswd设置密码（非交互式）
        echo "$username:$password" | sudo chpasswd
        echo -e "${GREEN}[成功] 用户 '$username' 创建成功！${NC}"
        write_log "成功创建用户: $username"
        id "$username"
    else
        echo -e "${RED}[错误] 用户创建失败！${NC}"
        return 1
    fi
}
```

#### **3.3.3 用户登录验证（支持三次重试）**

```bash
user_login() {
    echo -e "${BLUE}============================================${NC}"
    echo -e "${BLUE}           用户登录系统${NC}"
    echo -e "${BLUE}============================================${NC}"

    read -p "请输入用户名: " username

    # 检查用户是否存在
    if ! id "$username" &>/dev/null; then
        echo -e "${RED}[错误] 用户 '$username' 不存在！${NC}"
        return 1
    fi

    # 【安全机制】限制最多尝试3次
    local max_attempts=3
    local attempt=1

    while [ $attempt -le $max_attempts ]; do
        echo -e "${YELLOW}第 $attempt/$max_attempts 次尝试${NC}"
        read -s -p "请输入密码: " input_password
        echo ""

        # 【验证逻辑】使用su命令测试密码正确性
        echo "$input_password" | sudo -S su - "$username" \
            -c "echo 'success'" 2>/dev/null

        if [ $? -eq 0 ]; then
            echo -e "${GREEN}============================================${NC}"
            echo -e "${GREEN}       登录成功！欢迎回来，$username！${NC}"
            echo -e "${GREEN}============================================${NC}"
            # 显示用户详细信息
            echo "用户名: $username"
            echo "UID: $(id -u $username)"
            echo "主目录: $(grep "^$username:" /etc/passwd | cut -d: -f6)"
            write_log "用户 $username 成功登录系统"
            return 0
        else
            attempt=$((attempt + 1))
            if [ $attempt -le $max_attempts ]; then
                echo -e "${RED}[失败] 密码错误！还剩 $((max_attempts - attempt + 1)) 次机会${NC}"
            fi
        fi
    done

    # 超过最大尝试次数的处理
    echo -e "${RED}登录失败！已超过最大尝试次数(3次)${NC}"
    write_log "WARNING: 用户 $username 登录失败超过3次"
    return 1
}
```

create_folder() {
    echo -e "${BLUE}============================================${NC}"
    echo -e "${BLUE}           创建文件夹${NC}"
    echo -e "${BLUE}============================================${NC}"

    current_dir=$(pwd)
    echo -e "${YELLOW}当前工作目录: $current_dir${NC}"

    read -p "请输入要创建的文件夹路径: " folder_path

    # 输入验证
    if [ -z "$folder_path" ]; then
        echo -e "${RED}[错误] 文件夹路径不能为空！${NC}"
        return 1
    fi

    # 【核心逻辑】检查目录是否已存在
    if [ -d "$folder_path" ]; then
        echo -e "${RED}[错误] 文件夹 '$folder_path' 已存在！${NC}"

        return 1
    fi

    # 询问是否递归创建父目录
    read -p "是否需要自动创建不存在的父目录？(y/n): " create_parent

    if [ "$create_parent" = "y" ]; then
        # mkdir -p：递归创建所有不存在的父目录
        mkdir -p "$folder_path"
    else
        # 仅创建最后一级目录（父目录必须存在）
        mkdir "$folder_path"
    fi

    if [ $? -eq 0 ]; then
        # 设置权限为755：rwxr-xr-x
        # 所有者：读写执行；组和其他用户：读执行
        chmod 755 "$folder_path"
        echo -e "${GREEN}[成功] 文件夹创建成功！${NC}"
        ls -ld "$folder_path"  # 显示详细信息
    fi

delete_folder() {

    ARCHIVE_PATH="$TMP_DIR/apache-tomcat-${TOMCAT_VERSION}.tar.gz"

    # 解压到/opt目录
    sudo tar -xzf "$ARCHIVE_PATH" -C "$TOMCAT_INSTALL_DIR"

    # tar命令参数说明：
    # -x: 解压 (extract)
    # -z: 处理gzip压缩
    # -v: 显示过程 (verbose)
    # -f: 指定文件 (file)

    # 重命名目录
    EXTRACTED_DIR="${TOMCAT_INSTALL_DIR}/apache-tomcat-${TOMCAT_VERSION}"
    TOMCAT_HOME="${TOMCAT_INSTALL_DIR}/tomcat-${TOMCAT_VERSION}"
    [ ! -d "$TOMCAT_HOME" ] && sudo mv "$EXTRACTED_DIR" "$TOMCAT_HOME"

    # 设置权限
    sudo chown -R $USER:$USER "$TOMCAT_HOME"
    sudo chmod +x "$TOMCAT_HOME"/bin/*.sh

    # 创建软链接方便访问
    [ ! -L "/opt/tomcat" ] && sudo ln -s "$TOMCAT_HOME" /opt/tomcat

    echo -e "${GREEN}Tomcat安装完成: $TOMCAT_HOME${NC}"
}

check_tomcat_status() {
    PORT=${TOMCAT_PORT:-8080}

    echo -e "\n--- Tomcat运行状态 ---\n"

    # 检查进程
    if pgrep -f "org.apache.catalina.startup.Bootstrap" > /dev/null; then
        PID=$(pgrep -f "org.apache.catalina.startup.Bootstrap" | head -1)
        echo -e "  ${GREEN}● 进程运行中 (PID: $PID)${NC}"
    else
        echo -e "  ${RED}○ 进程未运行${NC}"
    fi

    # 检查端口监听
    if ss -tlnp 2>/dev/null | grep -q ":${PORT} "; then
        echo -e "  ${GREEN}● 端口 $PORT 监听中${NC}"
    fi

    # HTTP响应测试
    if command -v curl &>/dev/null; then
        HTTP_CODE=$(curl -s -o /dev/null -w "%{http_code}" http://localhost:$PORT/)
        [ "$HTTP_CODE" = "200" ] && echo -e "  ${GREEN}● HTTP正常 ($HTTP_CODE)${NC}"
    fi

    echo -e "\n访问地址: http://localhost:$PORT"
}

### **8.2 遇到的问题及解决方案**

| 问题类别 | 问题描述 | 解决方案 |
|----------|----------|----------|
| **权限不足** | 执行useradd等命令时提示Permission denied | 使用sudo提升权限；将用户加入sudo组 |
| **密码安全** | chpasswd明文传输密码不安全 | 在脚本中使用变量存储，避免硬编码；生产环境应使用密钥认证 |
| **中文乱码** | MySQL/Tomcat出现中文乱码 | 统一设置字符集为utf8mb4；配置URIEncoding="UTF-8" |
| **端口冲突** | Tomcat默认8080端口被占用 | 修改server.xml中的端口号；或结束占用进程 |
| **依赖缺失** | 缺少wget、curl等工具 | 提前安装：apt-get install wget curl |
| **网络问题** | 无法下载MySQL/Tomcat安装包 | 配置国内镜像源（如阿里云镜像）；或手动下载后离线安装 |
| **Java版本** | Tomcat要求的Java版本不匹配 | 安装兼容的JDK版本；设置正确的JAVA_HOME |
| **防火墙** | 外部无法访问Tomcat | 开放防火墙端口：firewall-cmd --add-port=8080/tcp |
