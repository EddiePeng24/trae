## 1. 题目一：用户管理系统

### 1.1 功能说明

实现用户的创建、密码修改、登录验证、用户删除等功能，提供交互式菜单界面。

### 1.2 核心代码

```bash
#!/bin/bash
# 题目1：用户管理系统

LOG_FILE="/var/log/user_management.log"
write_log() { echo "[$(date '+%Y-%m-%d %H:%M:%S')] $1" | sudo tee -a "$LOG_FILE" > /dev/null; }

# --- 创建用户 ---
create_user() {
    read -p "请输入用户名: " username
    [ -z "$username" ] && echo "用户名不能为空" && return 1
    id "$username" &>/dev/null && echo "用户已存在" && return 1
    read -s -p "请输入密码: " pwd; echo ""
    sudo useradd -m -s /bin/bash "$username"
    echo "$username:$pwd" | sudo chpasswd
    echo "用户 $username 创建成功"
    write_log "创建用户: $username"
}

# --- 修改密码 ---
change_password() {
    read -p "请输入用户名: " username
    id "$username" &>/dev/null || (echo "用户不存在" && return 1)
    read -s -p "请输入新密码: " pwd; echo ""
    echo "$username:$pwd" | sudo chpasswd
    echo "密码修改成功"
}

# --- 登录验证 ---
user_login() {
    read -p "请输入用户名: " username
    id "$username" &>/dev/null || (echo "用户不存在" && return 1)
    for i in 1 2 3; do
        read -s -p "密码 (尝试$i/3): " pwd; echo ""
        if echo "$pwd" | su -c "exit" "$username" &>/dev/null; then
            echo "登录成功，欢迎 $username"
            return 0
        fi
        echo "密码错误"
    done
    echo "连续3次错误，账号锁定"
    write_log "登录失败: $username"
    return 1
}

# --- 删除用户 ---
delete_user() {
    read -p "请输入要删除的用户名: " username
    [ "$username" = "root" ] && echo "禁止删除root" && return 1
    id "$username" &>/dev/null || (echo "用户不存在" && return 1)
    read -p "确认删除? (y/n): " confirm
    [ "$confirm" = "y" ] && sudo userdel -r "$username" && echo "删除成功" && write_log "删除用户: $username"
}

# --- 用户列表 ---
list_users() {
    echo "--- 系统普通用户 ---"
    awk -F: '$3>=1000 && $3!=65534 {printf "%-15s UID:%-6s HOME:%s\n", $1, $3, $6}' /etc/passwd
}

# --- 主菜单 ---
while true; do
    echo ""
    echo "==== 用户管理系统 ===="
    echo "1. 创建用户  2. 修改密码  3. 登录验证"
    echo "4. 删除用户  5. 用户列表  0. 退出"
    read -p "选择: " choice
    case $choice in
        1) create_user ;; 2) change_password ;;
        3) user_login ;;  4) delete_user ;;
        5) list_users ;;   0) echo "再见"; exit 0 ;;
        *) echo "无效选择" ;;
    esac
done
```

## 2. 题目二：文件操作系统

### 2.1 功能说明

实现文件夹/文件的创建、修改、删除、浏览等完整操作，提供安全保护机制。

### 2.2 核心代码

```bash
#!/bin/bash
# 题目2：文件操作系统

# --- 创建文件夹 ---
create_folder() {
    read -p "文件夹路径: " path
    [ -z "$path" ] && echo "路径不能为空" && return 1
    [ -d "$path" ] && echo "文件夹已存在" && return 1
    read -p "自动创建父目录? (y/n): " parent
    [ "$parent" = "y" ] && mkdir -p "$path" || mkdir "$path"
    chmod 755 "$path"
    echo "创建成功: $path"
    ls -ld "$path"
}

# --- 创建文件 ---
create_file() {
    read -p "文件路径: " path
    [ -z "$path" ] && echo "路径不能为空" && return 1
    read -p "写入初始内容? (y/n): " writeit
    if [ "$writeit" = "y" ]; then
        echo "请输入内容 (Ctrl+D 结束):"
        cat > "$path"
    else
        touch "$path"
    fi
    chmod 644 "$path"
    echo "文件创建成功"
}

# --- 修改文件 ---
modify_file() {
    echo "1. 追加内容  2. 覆盖内容  3. 字符串替换"
    read -p "选择: " mode
    read -p "文件路径: " path
    [ ! -f "$path" ] && echo "文件不存在" && return 1
    case $mode in
        1) echo "请输入追加内容:"; cat >> "$path" ;;
        2) echo "请输入新内容:"; cat > "$path" ;;
        3) read -p "查找: " old; read -p "替换: " new
           sed -i "s|$old|$new|g" "$path" ;;
        *) echo "无效选择"; return 1 ;;
    esac
    echo "修改完成"
}

# --- 删除文件/目录 ---
delete_file() {
    read -p "删除路径: " path
    [ ! -e "$path" ] && echo "路径不存在" && return 1
    case "$path" in /|/bin|/boot|/etc|/root) echo "禁止删除系统目录"; return 1 ;; esac
    echo "将要删除: $path"
    read -p "确认? (y/n): " confirm
    [ "$confirm" = "y" ] && rm -rf "$path" && echo "删除成功"
}

# --- 浏览目录 ---
browse_dir() {
    read -p "目录路径 (留空为当前): " path
    path=${path:-.}
    [ ! -d "$path" ] && echo "目录不存在" && return 1
    echo "--- $path ---"
    ls -lah "$path"
    echo ""
    echo "文件数: $(find "$path" -type f 2>/dev/null | wc -l)"
    echo "目录数: $(find "$path" -type d 2>/dev/null | wc -l)"
    echo "大小: $(du -sh "$path" 2>/dev/null | cut -f1)"
}

# --- 主菜单 ---
while true; do
    echo ""
    echo "==== 文件操作系统 ===="
    echo "1. 创建文件夹  2. 创建文件  3. 修改文件"
    echo "4. 删除         5. 浏览目录  0. 退出"
    read -p "选择: " choice
    case $choice in
        1) create_folder ;; 2) create_file ;;
        3) modify_file ;;   4) delete_file ;;
        5) browse_dir ;;    0) echo "再见"; exit 0 ;;
        *) echo "无效选择" ;;
    esac
done
```

## 3. 题目三：MySQL安装与配置

### 3.1 功能说明

自动检测环境、安装MySQL、配置参数、创建数据库和表、插入测试数据并验证连接。

### 3.2 核心代码

```bash
#!/bin/bash
# 题目3：MySQL安装与配置

MYSQL_ROOT_PASSWORD="Root@123456"

# --- 环境检测 ---
check_env() {
    echo "--- 环境检测 ---"
    echo "系统: $(cat /etc/os-release | grep "^NAME=" | cut -d'"' -f2)"
    echo "内存: $(free -m | awk '/Mem:/ {print $2 "MB"}')"
    echo "磁盘: $(df -BG / | awk 'NR==2 {print $4}')"
    if command -v mysql &>/dev/null; then
        echo "MySQL已安装: $(mysql --version 2>&1 | head -1)"
        return 0
    fi
    echo "MySQL未安装，准备安装..."
    return 1
}

# --- 安装MySQL ---
install_mysql() {
    echo "--- 安装MySQL ---"
    if command -v yum &>/dev/null; then
        sudo yum install -y https://dev.mysql.com/get/mysql80-community-release-el7-7.noarch.rpm 2>/dev/null
        sudo yum install -y mysql-community-server
    elif command -v apt &>/dev/null; then
        sudo apt-get update -qq && sudo apt-get install -y mysql-server
    fi
    if command -v mysql &>/dev/null; then
        echo "安装成功"
        sudo systemctl start mysqld 2>/dev/null || sudo systemctl start mysql
        sudo systemctl enable mysqld 2>/dev/null || sudo systemctl enable mysql
    else
        echo "安装失败" && return 1
    fi
}

# --- 创建数据库 ---
create_db() {
    echo "--- 创建数据库 student_management ---"
    sudo mysql <<'EOF'
CREATE DATABASE IF NOT EXISTS student_management CHARACTER SET utf8mb4;
USE student_management;

CREATE TABLE IF NOT EXISTS students (
    id INT PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(50) NOT NULL,
    gender ENUM('男','女') DEFAULT '男',
    age INT, phone VARCHAR(20), email VARCHAR(100)
);

CREATE TABLE IF NOT EXISTS courses (
    id INT PRIMARY KEY AUTO_INCREMENT,
    course_name VARCHAR(100) NOT NULL,
    credit DECIMAL(3,1)
);

CREATE TABLE IF NOT EXISTS scores (
    id INT PRIMARY KEY AUTO_INCREMENT,
    student_id INT NOT NULL, course_id INT NOT NULL,
    score DECIMAL(5,2), exam_date DATE,
    FOREIGN KEY (student_id) REFERENCES students(id),
    FOREIGN KEY (course_id) REFERENCES courses(id)
);
EOF
    echo "数据库和表创建完成"
}

# --- 插入测试数据 ---
insert_data() {

    sudo mysql student_management <<'EOF'

# --- 验证 ---
verify() {

# --- 主流程 ---

create_db
insert_data
verify
echo "完成！数据库: student_management  用户名: root  密码: $MYSQL_ROOT_PASSWORD"
```

## 4. 题目四：Tomcat安装与配置

### 4.1 功能说明

自动检测Java环境、下载Tomcat、解压安装、配置端口、启动服务并验证HTTP访问。

### 4.2 核心代码

```bash
#!/bin/bash
# 题目4：Tomcat安装与配置

TOMCAT_VERSION="10.1.18"
TOMCAT_HOME="/opt/tomcat-${TOMCAT_VERSION}"
TOMCAT_URL="https://dlcdn.apache.org/tomcat/tomcat-10/v${TOMCAT_VERSION}/bin/apache-tomcat-${TOMCAT_VERSION}.tar.gz"

# --- Java检测 ---
check_java() {
    echo "--- Java环境检测 ---"
    if command -v java &>/dev/null; then
        echo "已安装: $(java -version 2>&1 | head -1)"
        return 0
    fi
    echo "未检测到Java，开始安装..."
    if command -v apt &>/dev/null; then
        sudo apt-get install -y openjdk-11-jdk
    else
        sudo yum install -y java-11-openjdk
    fi
}

# --- 下载Tomcat ---
download_tomcat() {
    echo "--- 下载Tomcat $TOMCAT_VERSION ---"
    TMP="/tmp/tomcat_$$.tar.gz"
    if command -v wget &>/dev/null; then
        wget -q "$TOMCAT_URL" -O "$TMP"
    else
        curl -sL "$TOMCAT_URL" -o "$TMP"
    fi
    [ -s "$TMP" ] && echo "下载完成" || (echo "下载失败" && return 1)
}

# --- 安装配置 ---
install_tomcat() {
    echo "--- 安装Tomcat ---"
    sudo mkdir -p /opt
    sudo tar -xzf /tmp/tomcat_*.tar.gz -C /opt
    sudo ln -sfn "$TOMCAT_HOME" /opt/tomcat
    sudo chmod +x /opt/tomcat/bin/*.sh
    echo "安装完成: $TOMCAT_HOME"
    echo "快捷链接: /opt/tomcat"
}

# --- 配置管理员 ---
config_admin() {
    echo "--- 配置管理员 ---"
    sudo cp /opt/tomcat/conf/tomcat-users.xml /opt/tomcat/conf/tomcat-users.xml.bak
    sudo sed -i '/<\/tomcat-users>/i\
  <role rolename="manager-gui"/>\
  <user username="admin" password="admin123" roles="manager-gui"/>' \
    /opt/tomcat/conf/tomcat-users.xml
    echo "用户名: admin  密码: admin123"
}

# --- 启动 ---
start_tomcat() {
    echo "--- 启动Tomcat ---"
    sudo /opt/tomcat/bin/startup.sh
    sleep 3
    if pgrep -f tomcat &>/dev/null; then
        echo "启动成功"
    else
        echo "启动失败" && return 1
    fi
}

# --- HTTP验证 ---
verify_tomcat() {
    echo "--- HTTP访问验证 ---"
    sleep 2
    if command -v curl &>/dev/null; then
        CODE=$(curl -s -o /dev/null -w "%{http_code}" http://localhost:8080/)
        echo "状态码: $CODE"
        [ "$CODE" = "200" ] && echo "Tomcat正常运行" || echo "警告: 状态码异常"
    fi
    echo ""
    echo "========================================="
    echo "Tomcat首页: http://localhost:8080/"
    echo "管理界面: http://localhost:8080/manager/html"
    echo "用户名: admin   密码: admin123"
    echo "========================================="
}

# --- 主流程 ---
echo "==== Tomcat自动安装与配置 ===="
check_java
download_tomcat
install_tomcat
config_admin
start_tomcat
verify_tomcat
```

# --- 创建项目目录结构 ---
create_project() {
    echo "==== 创建项目: $PROJECT_NAME ===="
    sudo rm -rf "$PROJECT_DIR" 2>/dev/null
    sudo mkdir -p "$PROJECT_DIR/css" "$PROJECT_DIR/js" "$PROJECT_DIR/WEB-INF"
    create_index
    create_students
    create_about
    sudo chmod -R 755 "$PROJECT_DIR"
    echo "项目创建完成: $PROJECT_DIR"
}

# --- 启动服务 ---
start_services() {
    echo "==== 启动服务 ===="
    # MySQL
    if pgrep -x mysqld &>/dev/null || pgrep -x mysql &>/dev/null; then
        echo "[√] MySQL已运行"
    else
        sudo systemctl start mysqld 2>/dev/null || sudo systemctl start mysql
        echo "[√] MySQL已启动"
    fi
    # Tomcat
    if pgrep -f tomcat &>/dev/null; then
        echo "[√] Tomcat已运行，重启中..."
        sudo "$TOMCAT_HOME/bin/shutdown.sh" 2>/dev/null; sleep 2
    fi
    sudo "$TOMCAT_HOME/bin/startup.sh"
    sleep 3
    echo "[√] Tomcat已启动"
}

# --- 验证部署 ---
verify_deploy() {
    echo "==== 部署验证 ===="
    sleep 2
    local base="http://localhost:8080/$PROJECT_NAME"
    echo "验证页面: $base"
    for page in index.html students.html about.html; do
        if command -v curl &>/dev/null; then
            CODE=$(curl -s -o /dev/null -w "%{http_code}" "$base/$page")
            [ "$CODE" = "200" ] && echo "[√] $page ($CODE)" || echo "[×] $page ($CODE)"
        else
            echo "[?] $page"
        fi
    done
    echo ""
    echo "================================================"
    echo "部署完成！请访问:"
    echo "  首页: http://localhost:8080/$PROJECT_NAME/index.html"
    echo "  学生: http://localhost:8080/$PROJECT_NAME/students.html"
    echo "  关于: http://localhost:8080/$PROJECT_NAME/about.html"
    echo "================================================"
}

# --- 主流程 ---
echo "============================================"

check_env

create_project

start_services

verify_deploy

## 6. 总结与参考文献

### 6.1 项目总结

本课程设计完成了五个功能模块，形成一个完整的Linux系统管理自动化工具链：

| 模块 | 脚本文件 | 核心功能 | 技术要点 |
|------|---------|---------|---------|
| 用户管理 | script1_user_manage.sh | 用户增删改查、密码验证 | useradd/chpasswd/id/su |
| 文件操作 | script2_file_operation.sh | 文件/目录的增删改查 | mkdir/touch/sed/rm/ls |
| MySQL配置 | script3_mysql_config.sh | 自动安装、建库建表插数据 | mysql/DDL/DML语句 |
| Tomcat配置 | script4_tomcat_config.sh | Java检测、下载安装、启动验证 | tar/wget/curl/startup.sh |
| 项目部署 | script5_project_deploy.sh | 环境检测、创建项目、启动验证 | HTML/CSS/HTTP状态码 |

**五个脚本的协同关系：**

```
script1 (用户管理) ──┐
script2 (文件操作) ───┤
                     ├─→ script5 (项目部署)
script3 (MySQL配置) ──┤   整合前4项能力
script4 (Tomcat配置) ─┘   完成完整部署
```

### 6.2 关键技术点

1. **Shell脚本编程：** 变量、函数、条件判断、循环、管道与重定向
2. **Linux命令：** grep、sed、awk在文本处理中的应用
3. **系统管理：** useradd、chpasswd、systemctl等管理命令
4. **数据库操作：** 通过Here Document向MySQL传递SQL语句
5. **Web部署：** Tomcat目录结构、webapps部署模型
6. **HTTP验证：** 使用curl检测响应状态码
