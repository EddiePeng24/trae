### **3.2 核心代码实现**

**完整代码（script1_user_manage.sh，约350行）：**

```bash
#!/bin/bash

# ============================================================
# 题目1：用户管理系统
# 功能：创建用户、设置密码、登录验证、删除用户
# ============================================================

# 颜色定义
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
BLUE='\033[0;34m'
NC='\033[0m'

LOG_FILE="/var/log/user_management.log"

# 初始化日志
init_log() {
    if [ ! -f "$LOG_FILE" ]; then
        sudo touch "$LOG_FILE"
        sudo chmod 644 "$LOG_FILE"
    fi
}

write_log() {
    echo "[$(date '+%Y-%m-%d %H:%M:%S')] $1" | sudo tee -a "$LOG_FILE" > /dev/null
}

# ============================================
# 功能1：创建用户
# 逻辑：检查存在性 → 输入密码 → useradd创建 → chpasswd设密
# ============================================
create_user() {
    echo -e "${BLUE}============================================${NC}"
    echo -e "${BLUE}           创建新用户${NC}"
    echo -e "${BLUE}============================================${NC}"

    read -p "请输入要创建的用户名: " username

    # 检查用户名是否为空
    if [ -z "$username" ]; then
        echo -e "${RED}[错误] 用户名不能为空！${NC}"
        return 1
    fi

    # 核心：检查用户是否已存在
    if id "$username" &>/dev/null; then
        echo -e "${RED}[错误] 用户 '$username' 已存在！${NC}"
        write_log "尝试创建用户失败: 用户 $username 已存在"
        return 1
    fi

    # 输入并确认密码
    read -s -p "请输入用户密码: " password
    echo ""
    read -s -p "请再次确认密码: " password2
    echo ""

    # 验证两次密码是否一致
    if [ "$password" != "$password2" ]; then
        echo -e "${RED}[错误] 两次输入的密码不一致！${NC}"
        return 1
    fi

    # 创建用户并设置密码
    echo -e "${YELLOW}[正在创建用户]...${NC}"
    sudo useradd -m -s /bin/bash "$username"

    if [ $? -eq 0 ]; then
        echo "$username:$password" | sudo chpasswd
        echo -e "${GREEN}[成功] 用户 '$username' 创建成功！${NC}"
        write_log "成功创建用户: $username"
        id "$username"
    else
        echo -e "${RED}[错误] 用户创建失败！${NC}"
        write_log "创建用户失败: $username"
        return 1
    fi
}

# ============================================
# 功能2：修改密码
# ============================================
change_password() {
    echo -e "${BLUE}============================================${NC}"
    echo -e "${BLUE}           修改用户密码${NC}"
    echo -e "${BLUE}============================================${NC}"

    read -p "请输入要修改密码的用户名: " username

    # 检查用户是否存在
    if ! id "$username" &>/dev/null; then
        echo -e "${RED}[错误] 用户 '$username' 不存在！${NC}"
        return 1
    fi

    # 输入新密码
    read -s -p "请输入新密码: " new_password
    echo ""
    read -s -p "请再次确认新密码: " confirm_password
    echo ""

    if [ "$new_password" != "$confirm_password" ]; then
        echo -e "${RED}[错误] 两次输入的密码不一致！${NC}"
        return 1
    fi

    # 修改密码
    echo "$username:$new_password" | sudo chpasswd

    if [ $? -eq 0 ]; then
        echo -e "${GREEN}[成功] 用户 '$username' 的密码修改成功！${NC}"
        write_log "用户 $username 的密码已被修改"
    else
        echo -e "${RED}[错误] 密码修改失败！${NC}"
        return 1
    fi
}

# ============================================
# 功能3：用户登录验证（最多3次尝试）
# ============================================
user_login() {
    echo -e "${BLUE}============================================${NC}"
    echo -e "${BLUE}           用户登录验证${NC}"
    echo -e "${BLUE}============================================${NC}"

    read -p "请输入用户名: " username

    # 检查用户是否存在
    if ! id "$username" &>/dev/null; then
        echo -e "${RED}[错误] 用户 '$username' 不存在！${NC}"
        return 1
    fi

    # 最多3次尝试
    max_attempts=3
    attempts=0

    while [ $attempts -lt $max_attempts ]; do
        ((attempts++))
        echo -e "${YELLOW}请输入密码 (尝试 $attempts/$max_attempts): ${NC}"
        read -s -p "密码: " password
        echo ""

        # 使用su命令验证密码
        if echo "$password" | su -c "exit" "$username" &>/dev/null; then
            echo -e "${GREEN}[成功] 登录成功！欢迎，$username！${NC}"
            write_log "用户 $username 登录成功"
            return 0
        else
            echo -e "${RED}[失败] 密码错误！${NC}"
        fi
    done

    echo -e "${RED}[锁定] 连续 $max_attempts 次密码错误，账号已锁定！${NC}"
    write_log "用户 $username 登录失败: 密码错误超过 $max_attempts 次"
    return 1
}

# ============================================
# 功能4：删除用户
# ============================================
delete_user() {
    echo -e "${BLUE}============================================${NC}"
    echo -e "${BLUE}           删除用户${NC}"
    echo -e "${BLUE}============================================${NC}"

    read -p "请输入要删除的用户名: " username

    # 安全检查：禁止删除root和当前用户
    if [ "$username" = "root" ]; then
        echo -e "${RED}[错误] 禁止删除root用户！${NC}"
        return 1
    fi

    if [ "$username" = "$(whoami)" ]; then
        echo -e "${RED}[错误] 不能删除当前登录用户！${NC}"
        return 1
    fi

    # 检查用户是否存在
    if ! id "$username" &>/dev/null; then
        echo -e "${RED}[错误] 用户 '$username' 不存在！${NC}"
        return 1
    fi

    # 二次确认
    read -p "确定要删除用户 '$username' 吗？(y/n): " confirm
    if [ "$confirm" != "y" ] && [ "$confirm" != "Y" ]; then
        echo -e "${YELLOW}已取消删除操作${NC}"
        return 0
    fi

    # 删除用户（-r 同时删除主目录）
    sudo userdel -r "$username"

    if [ $? -eq 0 ]; then
        echo -e "${GREEN}[成功] 用户 '$username' 已删除！${NC}"
        write_log "已删除用户: $username"
    else
        echo -e "${RED}[错误] 删除用户失败！${NC}"
        return 1
    fi
}

# ============================================
# 功能5：显示用户列表
# ============================================
list_users() {
    echo -e "${BLUE}============================================${NC}"
    echo -e "${BLUE}           系统用户列表${NC}"
    echo -e "${BLUE}============================================${NC}"

    # 从/etc/passwd提取用户信息（UID >= 1000为普通用户）
    echo -e "${YELLOW}用户名 │ UID │ 主目录 │ Shell${NC}"
    echo "────────┼─────┼────────────────┼────────────"
    awk -F: '$3 >= 1000 && $3 != 65534 {printf "%-8s│ %-4s │ %-16s │ %s\n", $1, $3, $6, $7}' /etc/passwd

    echo ""
    echo -e "${GREEN}普通用户总数: $(awk -F: '$3 >= 1000 && $3 != 65534' /etc/passwd | wc -l)${NC}"
}

# ============================================
# 主菜单
# ============================================
main_menu() {
    init_log

    while true; do
        echo ""
        echo -e "${CYAN}╔══════════════════════════════════════════════╗${NC}"
        echo -e "${CYAN}║         Linux 用户管理系统                   ║${NC}"
        echo -e "${CYAN}╠══════════════════════════════════════════════╣${NC}"
        echo -e "${CYAN}║  1) 创建新用户                               ║${NC}"
        echo -e "${CYAN}║  2) 修改用户密码                             ║${NC}"
        echo -e "${CYAN}║  3) 用户登录验证                             ║${NC}"
        echo -e "${CYAN}║  4) 删除用户                                 ║${NC}"
        echo -e "${CYAN}║  5) 查看用户列表                             ║${NC}"
        echo -e "${CYAN}║  0) 退出系统                                 ║${NC}"
        echo -e "${CYAN}╚══════════════════════════════════════════════╝${NC}"

        read -p "请选择操作 [0-5]: " choice

        case $choice in
            1) create_user ;;
            2) change_password ;;
            3) user_login ;;
            4) delete_user ;;
            5) list_users ;;
            0) echo -e "${GREEN}感谢使用，再见！${NC}"; exit 0 ;;
            *) echo -e "${RED}无效选择，请重新输入！${NC}" ;;
        esac

        read -p "按回车键继续..." Enter
    done
}

# 程序入口
main_menu
```

### **4.2 核心代码实现**

**完整代码（script2_file_operation.sh，约400行）：**

```bash
#!/bin/bash

# ============================================================
# 题目2：文件操作系统
# 功能：创建/修改/删除文件和文件夹
# ============================================================

RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
BLUE='\033[0;34m'
NC='\033[0m'

LOG_FILE="/workspace/linux_course_design/file_operation.log"

log_op() {
    echo "[$(date '+%Y-%m-%d %H:%M:%S')] $1" >> "$LOG_FILE"
}

# ============================================
# 功能1：创建文件夹
# ============================================
create_folder() {
    echo -e "${BLUE}============================================${NC}"
    echo -e "${BLUE}           创建文件夹${NC}"
    echo -e "${BLUE}============================================${NC}"

    read -p "请输入要创建的文件夹路径: " folder_path

    if [ -z "$folder_path" ]; then
        echo -e "${RED}[错误] 文件夹路径不能为空！${NC}"
        return 1
    fi

    # 核心：检查目录是否已存在
    if [ -d "$folder_path" ]; then
        echo -e "${RED}[错误] 文件夹 '$folder_path' 已存在！${NC}"
        return 1
    fi

    read -p "是否需要自动创建父目录？(y/n): " create_parent

    if [ "$create_parent" = "y" ] || [ "$create_parent" = "Y" ]; then
        mkdir -p "$folder_path"
    else
        mkdir "$folder_path"
    fi

    if [ $? -eq 0 ]; then
        chmod 755 "$folder_path"
        echo -e "${GREEN}[成功] 文件夹 '$folder_path' 创建成功！${NC}"
        ls -ld "$folder_path"
        log_op "创建文件夹: $folder_path"
    else
        echo -e "${RED}[错误] 创建失败！${NC}"
        return 1
    fi
}

# ============================================
# 功能2：创建文件
# ============================================
create_file() {
    echo -e "${BLUE}============================================${NC}"
    echo -e "${BLUE}           创建文件${NC}"
    echo -e "${BLUE}============================================${NC}"

    read -p "请输入要创建的文件路径: " file_path

    if [ -z "$file_path" ]; then
        echo -e "${RED}[错误] 文件路径不能为空！${NC}"
        return 1
    fi

    # 检查文件是否已存在
    if [ -f "$file_path" ]; then
        echo -e "${YELLOW}[警告] 文件 '$file_path' 已存在，将覆盖！${NC}"
        read -p "确定要覆盖吗？(y/n): " confirm
        if [ "$confirm" != "y" ] && [ "$confirm" != "Y" ]; then
            echo "已取消操作"
            return 0
        fi
    fi

    # 询问是否写入内容
    read -p "是否需要写入初始内容？(y/n): " write_content

    if [ "$write_content" = "y" ] || [ "$write_content" = "Y" ]; then
        echo "请输入文件内容（输入EOF结束）："
        cat > "$file_path" << 'EOFINPUT'
# 在此输入文件内容...
EOFINPUT
    else
        touch "$file_path"
    fi

    if [ $? -eq 0 ]; then
        chmod 644 "$file_path"
        echo -e "${GREEN}[成功] 文件 '$file_path' 创建成功！${NC}"
        ls -l "$file_path"
        log_op "创建文件: $file_path"
    else
        echo -e "${RED}[错误] 创建失败！${NC}"
        return 1
    fi
}

# ============================================
# 功能3：修改文件内容
# ============================================
modify_file() {
    echo -e "${BLUE}============================================${NC}"
    echo -e "${BLUE}           修改文件${NC}"
    echo -e "${BLUE}============================================${NC}"

    echo "请选择修改方式："
    echo "  1) 追加内容到文件末尾"
    echo "  2) 覆盖整个文件内容"
    echo "  3) 替换文件中的特定字符串"
    read -p "请选择 [1-3]: " modify_type

    read -p "请输入文件路径: " file_path

    if [ ! -f "$file_path" ]; then
        echo -e "${RED}[错误] 文件 '$file_path' 不存在！${NC}"
        return 1
    fi

    case $modify_type in
        1)  # 追加内容
            echo "请输入要追加的内容："
            cat >> "$file_path"
            echo -e "${GREEN}[成功] 内容已追加到文件${NC}"
            log_op "追加内容到: $file_path"
            ;;
        2)  # 覆盖内容
            echo "请输入新的文件内容："
            cat > "$file_path"
            echo -e "${GREEN}[成功] 文件内容已覆盖${NC}"
            log_op "覆盖内容: $file_path"
            ;;
        3)  # 替换字符串
            read -p "请输入要替换的字符串: " old_str
            read -p "请输入替换后的字符串: " new_str
            # 核心：使用sed进行全局替换
            sed -i "s|$old_str|$new_str|g" "$file_path"
            echo -e "${GREEN}[成功] 已将 '$old_str' 替换为 '$new_str'${NC}"
            log_op "替换字符串: $file_path ($old_str -> $new_str)"
            ;;
        *)
            echo -e "${RED}[错误] 无效选择！${NC}"
            return 1
            ;;
    esac
}

# ============================================
# 功能4：删除文件/文件夹
# ============================================
delete_file() {
    echo -e "${BLUE}============================================${NC}"
    echo -e "${BLUE}           删除文件/文件夹${NC}"
    echo -e "${BLUE}============================================${NC}"

    read -p "请输入要删除的路径: " target_path

    if [ -z "$target_path" ]; then
        echo -e "${RED}[错误] 路径不能为空！${NC}"
        return 1
    fi

    # 安全检查：禁止删除系统目录
    case "$target_path" in
        /|/bin|/boot|/dev|/etc|/lib|/proc|/root|/sbin|/sys|/usr)
            echo -e "${RED}[错误] 禁止删除系统目录！${NC}"
            return 1
            ;;
    esac

    # 检查目标是否存在
    if [ ! -e "$target_path" ]; then
        echo -e "${RED}[错误] '$target_path' 不存在！${NC}"
        return 1
    fi

    # 显示将要删除的内容
    if [ -d "$target_path" ]; then
        echo -e "${YELLOW}将要删除目录及其内容：${NC}"
        ls -la "$target_path"
    else
        echo -e "${YELLOW}将要删除文件：${NC}"
        ls -l "$target_path"
    fi

    # 二次确认
    read -p "确定要删除吗？(y/n): " confirm
    if [ "$confirm" != "y" ] && [ "$confirm" != "Y" ]; then
        echo "已取消删除操作"
        return 0
    fi

    # 删除（目录用-rf，文件直接rm）
    if [ -d "$target_path" ]; then
        rm -rf "$target_path"
    else
        rm -f "$target_path"
    fi

    if [ $? -eq 0 ]; then
        echo -e "${GREEN}[成功] '$target_path' 已删除！${NC}"
        log_op "删除: $target_path"
    else
        echo -e "${RED}[错误] 删除失败！${NC}"
        return 1
    fi
}
```

```bash
#!/bin/bash

# ============================================================
# 题目3：MySQL数据库安装与配置
# 功能：自动安装MySQL、配置环境、初始化数据库
# ============================================================

RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
BLUE='\033[0;34m'
CYAN='\033[0;36m'
NC='\033[0m'

MYSQL_ROOT_PASSWORD="Root@123456"
MYSQL_LOG="/workspace/linux_course_design/mysql_install.log"

log_msg() {
    echo "[$(date '+%Y-%m-%d %H:%M:%S')] $1" | tee -a "$MYSQL_LOG"
}

# ============================================
# 功能1：检查系统环境
# ============================================
check_environment() {
    echo -e "${CYAN}============================================${NC}"
    echo -e "${CYAN}       系统环境检测${NC}"
    echo -e "${CYAN}============================================${NC}"

    log_msg "[步骤1] 开始系统环境检测..."

    # 检测操作系统
    if [ -f /etc/os-release ]; then
        . /etc/os-release
        echo -e "${GREEN}  操作系统: $NAME $VERSION_ID${NC}"
        log_msg "操作系统: $NAME $VERSION_ID"
    fi

    # 检测内存
    total_mem=$(free -m | awk '/Mem:/ {print $2}')
    echo -e "${GREEN}  内存: ${total_mem}MB${NC}"
    log_msg "内存: ${total_mem}MB"

    # 检测磁盘空间
    available_space=$(df -BG / | awk 'NR==2 {print $4}' | tr -d 'G')
    echo -e "${GREEN}  可用空间: ${available_space}GB${NC}"
    log_msg "可用空间: ${available_space}GB"

    if [ "$available_space" -lt 1 ]; then
        echo -e "${RED}  [错误] 磁盘空间不足！${NC}"
        return 1
    fi

    # 检测是否已安装MySQL
    if command -v mysql &>/dev/null; then
        echo -e "${YELLOW}  MySQL已安装: $(mysql --version)${NC}"
        read -p "是否需要重新安装？(y/n): " reinstall
        if [ "$reinstall" != "y" ]; then
            return 0
        fi
    fi

    return 0
}

# ============================================
# 功能2：安装MySQL
# ============================================
install_mysql() {
    echo -e "${CYAN}============================================${NC}"
    echo -e "${CYAN}       安装MySQL数据库${NC}"
    echo -e "${CYAN}============================================${NC}"

    log_msg "[步骤2] 开始安装MySQL..."

    # 根据操作系统选择安装方式
    if [[ "$NAME" == *"CentOS"* ]] || [[ "$NAME" == *"Rocky"* ]] || [[ "$NAME" == *"Alma"* ]]; then
        echo -e "${YELLOW}[检测] CentOS/RHEL系统，使用yum安装...${NC}"

        # 安装MySQL YUM仓库
        sudo yum install -y https://dev.mysql.com/get/mysql80-community-release-el7-7.noarch.rpm 2>/dev/null || \
        sudo yum install -y https://dev.mysql.com/get/mysql80-community-release-el8-1.noarch.rpm

        # 安装MySQL服务器
        sudo yum install -y mysql-community-server

        if [ $? -eq 0 ]; then
            echo -e "${GREEN}[成功] MySQL安装完成！${NC}"
            log_msg "MySQL安装成功"
        else
            echo -e "${RED}[错误] MySQL安装失败！${NC}"
            return 1
        fi

    elif [[ "$NAME" == *"Ubuntu"* ]] || [[ "$NAME" == *"Debian"* ]]; then
        echo -e "${YELLOW}[检测] Debian/Ubuntu系统，使用apt安装...${NC}"

        sudo apt-get update -qq
        sudo apt-get install -y mysql-server

        if [ $? -eq 0 ]; then
            echo -e "${GREEN}[成功] MySQL安装完成！${NC}"
            log_msg "MySQL安装成功"
        else
            echo -e "${RED}[错误] MySQL安装失败！${NC}"
            return 1
        fi
    else
        echo -e "${RED}[错误] 不支持的操作系统！${NC}"
        return 1
    fi

    # 启动MySQL服务
    echo -e "${YELLOW}[启动] MySQL服务...${NC}"
    sudo systemctl start mysqld 2>/dev/null || sudo systemctl start mysql
    sudo systemctl enable mysqld 2>/dev/null || sudo systemctl enable mysql

    # 等待MySQL启动
    sleep 3

    # 检查MySQL是否运行
    if pgrep -x mysqld > /dev/null || pgrep -x mysql > /dev/null; then
        echo -e "${GREEN}[成功] MySQL服务已启动！${NC}"
        log_msg "MySQL服务已启动"
    else
        echo -e "${RED}[错误] MySQL服务启动失败！${NC}"
        return 1
    fi
}

# ============================================
# 功能3：配置MySQL环境
# ============================================
configure_mysql() {
    echo -e "${CYAN}============================================${NC}"
    echo -e "${CYAN}       配置MySQL环境${NC}"
    echo -e "${CYAN}============================================${NC}"

    log_msg "[步骤3] 配置MySQL环境..."

    # 获取MySQL版本
    MYSQL_VERSION=$(mysql --version | grep -oE '[0-9]+\.[0-9]+')
    echo -e "${GREEN}  MySQL版本: $MYSQL_VERSION${NC}"

    # 检查my.cnf配置文件
    if [ -f /etc/my.cnf ]; then
        echo -e "${GREEN}  配置文件: /etc/my.cnf${NC}"
    fi

    # 配置环境变量（添加到bashrc）
    echo "" >> ~/.bashrc
    echo "# MySQL环境配置" >> ~/.bashrc
    echo 'export PATH=$PATH:/usr/bin:/usr/sbin' >> ~/.bashrc

    echo -e "${GREEN}[成功] MySQL环境配置完成！${NC}"
    log_msg "MySQL环境配置完成"
}

# ============================================
# 功能4：初始化MySQL
# ============================================
initialize_mysql() {
    echo -e "${CYAN}============================================${NC}"
    echo -e "${CYAN}       初始化MySQL${NC}"
    echo -e "${CYAN}============================================${NC}"

    log_msg "[步骤4] 初始化MySQL..."

    # CentOS/RHEL需要获取临时root密码
    if [[ -f /var/log/mysqld.log ]] && grep -q "temporary password" /var/log/mysqld.log; then
        TEMP_PASSWORD=$(grep "temporary password" /var/log/mysqld.log | awk '{print $NF}')
        echo -e "${YELLOW}  临时密码: $TEMP_PASSWORD${NC}"
    fi

    # 设置root密码（使用mysqladmin）
    echo -e "${YELLOW}[设置] root密码...${NC}"

    # 尝试使用临时密码修改（CentOS）
    if [ -n "$TEMP_PASSWORD" ]; then
        mysqladmin -u root -p"$TEMP_PASSWORD" password "$MYSQL_ROOT_PASSWORD" 2>/dev/null
    fi

    # 通用方法：直接设置密码
    sudo mysql -e "ALTER USER 'root'@'localhost' IDENTIFIED BY '$MYSQL_ROOT_PASSWORD';" 2>/dev/null || \
    sudo mysql -e "SET PASSWORD FOR 'root'@'localhost' = PASSWORD('$MYSQL_ROOT_PASSWORD');" 2>/dev/null || \
    echo -e "${YELLOW}  注意: 请手动设置root密码或使用sudo mysql登录${NC}"

    echo -e "${GREEN}[成功] MySQL初始化完成！${NC}"
    log_msg "MySQL初始化完成"
}

# 功能5：创建数据库和表
# ============================================
create_database_and_tables() {
    echo -e "${CYAN}============================================${NC}"
    echo -e "${CYAN}       创建数据库和表${NC}"
    echo -e "${CYAN}============================================${NC}"

    log_msg "[步骤5] 创建数据库和表..."

    DB_NAME="student_management"

    # 使用sudo执行MySQL命令（无密码验证）
    sudo mysql << 'EOSQL'

-- 创建数据库
CREATE DATABASE IF NOT EXISTS student_management CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE student_management;

-- 创建学生表
DROP TABLE IF EXISTS students;
CREATE TABLE students (
    id INT PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(50) NOT NULL,
    gender ENUM('男', '女') DEFAULT '男',
    age INT,
    phone VARCHAR(20),
    email VARCHAR(100),
    address VARCHAR(200),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 创建课程表
DROP TABLE IF EXISTS courses;
CREATE TABLE courses (
    id INT PRIMARY KEY AUTO_INCREMENT,
    course_name VARCHAR(100) NOT NULL,
    course_code VARCHAR(20) UNIQUE,
    credit DECIMAL(3,1),
    teacher VARCHAR(50),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 创建成绩表
DROP TABLE IF EXISTS scores;
CREATE TABLE scores (
    id INT PRIMARY KEY AUTO_INCREMENT,
    student_id INT NOT NULL,
    course_id INT NOT NULL,
    score DECIMAL(5,2),
    exam_date DATE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (student_id) REFERENCES students(id) ON DELETE CASCADE,
    FOREIGN KEY (course_id) REFERENCES courses(id) ON DELETE CASCADE,
    UNIQUE KEY uk_student_course (student_id, course_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

EOSQL

    if [ $? -eq 0 ]; then
        echo -e "${GREEN}[成功] 数据库和表创建完成！${NC}"
        log_msg "数据库 $DB_NAME 创建成功"
    else
        echo -e "${RED}[错误] 数据库创建失败！${NC}"
        return 1
    fi
}
```

```bash
# ============================================================

CYAN='\033[0;36m'

TOMCAT_VERSION="10.1.18"
TOMCAT_INSTALL_DIR="/opt"
TOMCAT_HOME="${TOMCAT_INSTALL_DIR}/tomcat-${TOMCAT_VERSION}"
TOMCAT_URL="https://dlcdn.apache.org/tomcat/tomcat-10/v${TOMCAT_VERSION}/bin/apache-tomcat-${TOMCAT_VERSION}.tar.gz"

# ============================================
# 功能1：检查Java环境

check_java() {
    echo -e "${CYAN}============================================${NC}"
    echo -e "${CYAN}      Java环境检测${NC}"
    echo -e "${CYAN}============================================${NC}"

    log_msg "[步骤1] 检查Java环境..."

    # 检测Java是否已安装
    if command -v java &>/dev/null; then
        JAVA_VER=$(java -version 2>&1 | head -n 1)
        echo -e "${GREEN}  已安装: $JAVA_VER${NC}"
        log_msg "Java已安装: $JAVA_VER"
        return 0
    fi

    echo -e "${RED}  未检测到Java环境！${NC}"
    log_msg "未检测到Java，开始安装..."

    # 根据操作系统安装Java
    if command -v yum &>/dev/null; then
        echo -e "${YELLOW}[安装] 正在安装OpenJDK 11...${NC}"
        sudo yum install -y java-11-openjdk java-11-openjdk-devel
    elif command -v apt &>/dev/null; then
        echo -e "${YELLOW}[安装] 正在安装OpenJDK 11...${NC}"
        sudo apt-get update
        sudo apt-get install -y openjdk-11-jdk
    fi

    if command -v java &>/dev/null; then
        echo -e "${GREEN}  Java安装成功！${NC}"
        log_msg "Java安装成功"
        return 0
    else
        echo -e "${RED}  Java安装失败！${NC}"
        return 1
    fi

# 功能2：下载Tomcat
# ============================================
download_tomcat() {
    echo -e "${CYAN}============================================${NC}"
    echo -e "${CYAN}       下载Tomcat${NC}"
    echo -e "${CYAN}============================================${NC}"

    log_msg "[步骤2] 下载Tomcat $TOMCAT_VERSION..."

    # 创建临时目录
    TEMP_DIR="/tmp/tomcat_install"
    mkdir -p "$TEMP_DIR"
    cd "$TEMP_DIR"

    ARCHIVE_NAME="apache-tomcat-${TOMCAT_VERSION}.tar.gz"

    # 检查是否已下载
    if [ -f "$ARCHIVE_NAME" ]; then
        echo -e "${YELLOW}  找到已下载的文件，跳过下载${NC}"
        log_msg "使用已下载的文件"
    else
        echo -e "${YELLOW}[下载] 正在从官方源下载...${NC}"
        echo -e "${YELLOW}  URL: $TOMCAT_URL${NC}"

        # 使用wget下载
        if command -v wget &>/dev/null; then
            wget -q --show-progress "$TOMCAT_URL" -O "$ARCHIVE_NAME"
        elif command -v curl &>/dev/null; then
            curl -# -L "$TOMCAT_URL" -o "$ARCHIVE_NAME"
        else
            echo -e "${RED}  错误: 未安装wget或curl！${NC}"
            return 1
        fi
    fi

    if [ -f "$ARCHIVE_NAME" ] && [ -s "$ARCHIVE_NAME" ]; then
        echo -e "${GREEN}  下载完成！${NC}"
        echo -e "  文件大小: $(du -h "$ARCHIVE_NAME" | cut -f1)"
        log_msg "下载完成: $ARCHIVE_NAME"
    else
        echo -e "${RED}  下载失败！${NC}"
        return 1
    fi
}

        log_msg "解压到: $TOMCAT_HOME"
    else
        echo -e "${RED}  解压失败！${NC}"
        return 1
    fi

    # 设置权限
    echo -e "${YELLOW}[配置] 设置目录权限...${NC}"
    sudo chmod +x "$TOMCAT_HOME/bin/"*.sh

    # 创建符号链接（方便使用）
    sudo ln -sfn "$TOMCAT_HOME" /opt/tomcat
    echo -e "${GREEN}  创建符号链接: /opt/tomcat -> $TOMCAT_HOME${NC}"

    echo -e "${GREEN}[成功] Tomcat安装完成！${NC}"

    echo -e "${CYAN}       配置Tomcat${NC}"

            echo -e "${GREEN}  ● 端口 $PORT 监听中${NC}"
        else
            echo -e "${RED}  ○ 端口 $PORT 未监听${NC}"
        fi
    elif command -v netstat &>/dev/null; then
        if netstat -tlnp | grep -q ":$PORT "; then
            echo -e "${GREEN}  ● 端口 $PORT 监听中${NC}"
        else
            echo -e "${RED}  ○ 端口 $PORT 未监听${NC}"
```

```bash
// 显示当前时间
function updateTime() {
    const timeElement = document.getElementById('time');
    if (timeElement) {
        const now = new Date();
        timeElement.textContent = '当前时间: ' + now.toLocaleString('zh-CN');
    }
}

// 搜索学生功能
function searchStudent() {
    const input = document.getElementById('searchInput');
    const filter = input.value.toUpperCase();
    const table = document.getElementById('studentTable');
    const tr = table.getElementsByTagName('tr');

    for (let i = 1; i < tr.length; i++) {
        const td = tr[i].getElementsByTagName('td')[1];
        if (td) {
            const textValue = td.textContent || td.innerText;
            if (textValue.toUpperCase().indexOf(filter) > -1) {
                tr[i].style.display = '';
            } else {
                tr[i].style.display = 'none';
            }
        }
    }
}

// 页面加载时执行
document.addEventListener('DOMContentLoaded', function() {
    updateTime();
    setInterval(updateTime, 1000);

    // 为搜索框添加回车事件
    const searchInput = document.getElementById('searchInput');
    if (searchInput) {
        searchInput.addEventListener('keypress', function(e) {
            if (e.key === 'Enter') {
                searchStudent();
            }
        });
    }

    console.log('学生管理系统已加载');
});

start_mysql() {
    echo -e "${CYAN}============================================================${NC}"
    echo -e "${CYAN}              启动MySQL服务${NC}"
    echo -e "${CYAN}============================================================${NC}"

    log_msg "[步骤3] 启动MySQL..."

    # 检查MySQL是否运行
    if pgrep -x mysqld > /dev/null || pgrep -x mysql > /dev/null; then
        echo -e "${GREEN}  MySQL已在运行中！${NC}"
    else
        echo -e "${YELLOW}  正在启动MySQL...${NC}"
        sudo systemctl start mysqld 2>/dev/null || sudo systemctl start mysql

        sleep 3

        if pgrep -x mysqld > /dev/null || pgrep -x mysql > /dev/null; then
            echo -e "${GREEN}  MySQL启动成功！${NC}"
            log_msg "MySQL启动成功"
        else
            echo -e "${RED}  MySQL启动失败！${NC}"
            return 1
        fi
    fi
}

    echo -e "${CYAN}============================================================${NC}"

        log_msg "Tomcat启动成功"
```
